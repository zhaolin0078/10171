import pygame import math import random import sys # 初始化 Pygame pygame.init() pygame.mixer.init() # 初始化音频模块 info = pygame.display.Info() WIDTH, HEIGHT = info.current_w, info.current_h screen = pygame.display.set_mode((WIDTH, HEIGHT), pygame.FULLSCREEN) pygame.display.set_caption("跳动的蓝心") clock = pygame.time.Clock() class Particle: def __init__(self): self.reset() def reset(self): self.t = random.uniform(0, math.pi * 2) self.is_inner = random.random() > 0.35 self.offset = random.uniform(0, 0.85) if self.is_inner else random.uniform(0.85, 1.1) self.base_size = random.uniform(1.5, 3.0) if self.is_inner else random.uniform(3.0, 6.0) self.phase = random.uniform(0, math.pi * 2) if self.is_inner: self.color = (50, 150, 255) else: self.color = (120, 230, 255) self.trail = [] def draw(self, surface, time_val): x_base = 16 * (math.sin(self.t) ** 3) y_base = -(13 * math.cos(self.t) - 5 * math.cos(2 * self.t) - 2 * math.cos(3 * self.t) - math.cos(4 * self.t)) scale = (HEIGHT / 800) * 13 pulse = 1 + math.sin(time_val * 3) * 0.06 noise_x = math.sin(time_val * 2 + self.phase) * 5 noise_y = math.cos(time_val * 2 + self.phase) * 5 x = WIDTH / 2 + (x_base * self.offset * scale * pulse) + noise_x y = HEIGHT / 2 + (y_base * self.offset * scale * pulse) + noise_y self.trail.append((x, y)) if len(self.trail) > 4: self.trail.pop(0) for i, (tx, ty) in enumerate(self.trail): trail_alpha = (i / len(self.trail)) * 100 r = self.base_size * (i / len(self.trail)) if r > 0.5: pygame.draw.circle(surface, (*self.color, trail_alpha), (int(tx), int(ty)), int(r)) alpha = 0.6 + math.sin(time_val * 4 + self.phase) * 0.4 r_size = int(self.base_size * alpha) if r_size > 0: cr = min(255, int(self.color[0] * alpha * 1.5)) cg = min(255, int(self.color[1] * alpha * 1.5)) cb = min(255, int(self.color[2] * alpha * 1.5)) pygame.draw.circle(surface, (cr, cg, cb), (int(x), int(y)), r_size) def main(): # 加载背景音乐（请确保 music.mp3 和代码在同一个文件夹） try: pygame.mixer.music.load("letter.mp3") pygame.mixer.music.set_volume(0.6) # 音量设置 60% pygame.mixer.music.play(-1) # 无限循环播放 except Exception as e: print("⚠️ 找不到 music.mp3！请确保你把文件改名并放在了代码同一个文件夹。") print("报错信息：", e) particle_surface = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) particles = [Particle() for _ in range(2000)] start_time = pygame.time.get_ticks() / 1000.0 running = True while running: for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: running = False elif event.type == pygame.MOUSEBUTTONDOWN: running = False current_time = (pygame.time.get_ticks() / 1000.0) - start_time particle_surface.fill((0, 0, 0, 0)) screen.fill((5, 5, 10)) for p in particles: p.draw(particle_surface, current_time) screen.blit(particle_surface, (0, 0), special_flags=pygame.BLEND_RGBA_ADD) pygame.display.flip() clock.tick(60) pygame.quit() sys.exit() if __name__ == "__main__": main()
